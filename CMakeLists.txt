cmake_minimum_required(VERSION 3.20)
project(event_lob VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LOB_ASAN_UBSAN "Enable address and undefined behavior sanitizers" OFF)
option(LOB_TSAN "Enable thread sanitizer" OFF)
if(LOB_ASAN_UBSAN AND LOB_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
if(LOB_ASAN_UBSAN OR LOB_TSAN)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizer configuration requires Clang or GCC")
  endif()
  add_compile_options(-fno-omit-frame-pointer -g)
  if(LOB_ASAN_UBSAN)
    add_compile_options(-fsanitize=address,undefined)
    add_link_options(-fsanitize=address,undefined)
  else()
    add_compile_options(-fsanitize=thread)
    add_link_options(-fsanitize=thread)
  endif()
endif()
find_package(Threads REQUIRED)
add_library(lob src/engine.cpp src/persistence.cpp)
target_include_directories(lob PUBLIC include)
target_compile_options(lob PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(lob PUBLIC Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(lob_tests tests/tests.cpp)
  target_include_directories(lob_tests SYSTEM PRIVATE third_party)
  target_link_libraries(lob_tests PRIVATE lob)
  add_test(NAME correctness COMMAND lob_tests)
endif()
add_executable(lob_demo apps/demo.cpp)
target_link_libraries(lob_demo PRIVATE lob)
add_executable(lob_bench apps/benchmark.cpp)
target_link_libraries(lob_bench PRIVATE lob)
